cmake_minimum_required(VERSION 3.22)
project(GefRecentCodecTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
include(../../cmake/RecentCompressors.cmake)
add_executable(benchmark_input_test benchmark_input.cpp)
target_link_libraries(benchmark_input_test PRIVATE gef_benchmark_input)
# Exercise the same unsafe caller flags that exposed the original regression.
target_compile_options(benchmark_input_test PRIVATE -O3 -ffast-math)
add_test(NAME benchmark_input COMMAND benchmark_input_test)
add_executable(recent_harness_test harness.cpp)
target_link_libraries(recent_harness_test PRIVATE gef_recent_compressors)
add_test(NAME recent_harness COMMAND recent_harness_test)
add_executable(recent_warning_policy warning_policy.cpp)
target_compile_options(recent_warning_policy PRIVATE -fno-fast-math -ffp-contract=off)
